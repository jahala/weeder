+/**
+ * The casting ledger — spend divided by PROVEN work.
+ *
+ * Two feeds prove the class. The REAL spike journal (test/fixtures/journal.jsonl,
+ * the pdf-unifier build, vendored verbatim) is what the runner actually writes
+ * today: duration and attempts, no tokens, no provider — so the ledger must
+ * count that spend without inventing an owner for it. The synthetic journals
+ * carry the additive `provider`/`model`/`telemetry.tokens` fields pleach
+ * accepted, which is how attribution behaves once they land. Every yield
+ * assertion pairs with its adversarial counter-check: a provider that burns
+ * nothing and proves nothing must never out-rank one that pays and delivers.
+ */
+
+import { describe, it, expect } from 'vitest';
+import { mkdirSync, mkdtempSync, readFileSync, rmSync, writeFileSync } from 'node:fs';
+import { dirname, join } from 'node:path';
+import { tmpdir } from 'node:os';
+import { spawnSync } from 'node:child_process';
+import { fileURLToPath } from 'node:url';
+import { main } from '../src/cli/index.js';
+import { parseJournal, type JournalEvent } from '../src/watch/watch.js';
+import { parse } from '../src/core/parse.js';
+import type { CheckMethod } from '../src/core/types.js';
+import { castingLedger, claimKindsOf, rankCast, castFor, renderCasting, type ClaimIndex } from '../src/run/casting.js';
+
+const here = dirname(fileURLToPath(import.meta.url));
+const realJournal = readFileSync(join(here, 'fixtures', 'journal.jsonl'), 'utf8');
+
+/** Build a journal the way the reader sees one — through the real line parser. */
+const journal = (events: Array<Record<string, unknown>>): JournalEvent[] =>
+  parseJournal(events.map((e) => JSON.stringify(e)).join('\n'));
+
+const claims = (entries: Record<string, CheckMethod[]>): ClaimIndex => new Map(Object.entries(entries));
+
+const verdict = (node: string, over: Record<string, unknown> = {}): Record<string, unknown> =>
+  ({ event: 'verdict', node, status: 'done', attempts: 1, durationMs: 1000, telemetry: {}, ...over });
+
+const closed = (node: string): Record<string, unknown> => ({ event: 'closed', node, sha: `${node}-sha` });
+
+const cellOf = (ledger: ReturnType<typeof castingLedger>, provider: string | null, kind: string) =>
+  ledger.cells.find((c) => c.provider === provider && c.kind === kind);
+
+const kindsOf = (ledger: ReturnType<typeof castingLedger>): string[] => ledger.cells.map((c) => c.kind).sort();

+    expect(ledger.totals.durationMs).toBe(14_359_031);
+    expect(ledger.totals.attempts).toBe(15);
+    expect(ledger.totals.closes).toBe(7);
+    expect(ledger.totals.verifiedClaims).toBe(7);
+    expect(ledger.totals.msPerVerifiedClaim).toBeCloseTo(14_359_031 / 7, 6);
+  });
+
+  it('names no rate it cannot earn: this runner reports no tokens, so there is no token rate', () => {
+    expect(ledger.totals.tokens).toBe(0);
+    expect(ledger.totals.tokenReports).toBe(0);
+    expect(ledger.totals.tokensPerVerifiedClaim).toBeNull();
+  });

+  it('counts unattributed spend without folding it into a named provider', () => {
+    expect(ledger.totals.attributedDispatches).toBe(0);
+    expect(ledger.cells).toHaveLength(1);
+    expect(ledger.cells[0]!.provider).toBeNull();

+describe('unverified work never counts as yield', () => {

+  it('published-but-unverified is spend, not yield — tend declining to close is a failure to prove', () => {

+  it('blocked and dead dispatches are spend too — every attempt is paid for', () => {

+});

+    verdict('d', { provider: 'flaky', model: 'k-1', durationMs: 50_000 }), closed('d'),

+/**
+ * The wiring: the ledger is not a library nobody calls. `tend2 cast` reads the
+ * journal beside a map and prints it; `emit-plan --cast auto` casts the plan
+ * from the same numbers.
+ */
+describe('the CLI surfaces the ledger', () => {
+  const withRepo = (journalText: string | null, body: (repo: string, map: string) => void): void => {
+    const repo = mkdtempSync(join(tmpdir(), 'tend2-cast-cli-'));
+    try {
+      spawnSync('git', ['init', '-q'], { cwd: repo });
+      const map = join(repo, 'docs', 'tend2');
+      mkdirSync(map, { recursive: true });
+      if (journalText !== null) {
+        mkdirSync(join(repo, '.git', 'pleach'), { recursive: true });
+        writeFileSync(join(repo, '.git', 'pleach', 'journal.jsonl'), journalText);
+      }
+      body(repo, map);
+    } finally {
+      rmSync(repo, { recursive: true, force: true });
+    }
+  };
+
+  const capture = (run: () => number): { code: number; out: string } => {
+    const logs: string[] = [];
+    const orig = console.log;
+    console.log = (msg: string) => { logs.push(String(msg)); };
+    try {
+      return { code: run(), out: logs.join('\n') };
+    } finally {
+      console.log = orig;
+    }
+  };
+
+  const loopFile = (map: string, id: string, checks: string): void => {
+    writeFileSync(join(map, `${id}.tend2.html`),
+      `<!doctype html>\n<title>${id} · t</title>\n<script type="text/markdown" id="loop">\n# ${id} — d\n**Goal.** g.\n\n## Tests\n${checks}\n\n## Tried\n- 2026-08-19 seed\n</script>\n`);
+  };
+
+  it('`tend2 cast` prints the ledger for the map beside the journal', () => {
+    withRepo(realJournal, (repo, map) => {
+      loopFile(map, 'pdf-unifier', '- [x] (code) merges · t.ts @abc1234');
+      const { code, out } = capture(() => main(['cast', map]));
+      expect(code).toBe(0);
+      expect(out).toContain('casting ledger');
+      expect(out).toContain('proven claim');
+    });
+  });
+
+  it('`tend2 cast` says so plainly when no run has ever been journaled', () => {
+    withRepo(null, (repo, map) => {
+      const { code, out } = capture(() => main(['cast', map]));
+      expect(code).toBe(0);
+      expect(out).toContain('no pleach journal');
+    });
+  });
+
+  it('`emit-plan --cast auto` casts the plan from the journal beside the map', () => {
+    const priors = [
+      { event: 'verdict', node: 'done-cheap', provider: 'cheap', model: 'c-1', status: 'done', attempts: 1, durationMs: 100_000, telemetry: {} },
+      { event: 'closed', node: 'done-cheap', sha: 'aaa' },
+      { event: 'verdict', node: 'done-dear', provider: 'dear', model: 'd-1', status: 'done', attempts: 1, durationMs: 400_000, telemetry: {} },
+      { event: 'closed', node: 'done-dear', sha: 'bbb' },
+    ].map((e) => JSON.stringify(e)).join('\n');
+
+    withRepo(priors, (repo, map) => {
+      loopFile(map, 'done-cheap', '- [x] (code) proven · t.ts @abc1234');
+      loopFile(map, 'done-dear', '- [x] (code) proven · t.ts @abc1234');
+      loopFile(map, 'open', '- [ ] (code) todo · t.ts');
+      const { code, out } = capture(() => main(['emit-plan', map, '--repo-root', repo, '--cast', 'auto']));
+      expect(code).toBe(0);
+      const plan = JSON.parse(out) as { nodes: Array<{ id: string; worker?: { provider?: string }; accept?: { audit?: { provider: string } } }> };
+      const node = plan.nodes.find((n) => n.id === 'open')!;
+      expect(node.worker?.provider).toBe('cheap');
+      expect(node.accept?.audit?.provider).toBe('dear');
+    });
+  });
+
+  it('`emit-plan --cast auto` refuses rather than inventing a cast when nothing has run', () => {
+    withRepo(null, (repo, map) => {
+      loopFile(map, 'open', '- [ ] (code) todo · t.ts');
+      const errs: string[] = [];
+      const orig = console.error;
+      console.error = (msg: string) => { errs.push(String(msg)); };
+      try {
+        expect(main(['emit-plan', map, '--repo-root', repo, '--cast', 'auto'])).toBe(1);
+        expect(errs.join('\n')).toContain('nothing has run yet');
+      } finally {
+        console.error = orig;
+      }
+    });
+  });
